{%- comment -%} Outputs a prose block from _copy/.md, or from _copy/zh-tw/.md on the Chinese pages (falling back to English if that translation doesn't exist yet). Usage: {% include copy.html name="intro" %} {%- endcomment -%} {%- include lang.html -%} {%- assign block = nil -%} {%- if zh -%} {%- assign copy_path = "_copy/zh-tw/" | append: include.name | append: ".md" -%} {%- assign block = site.copy | where: "relative_path", copy_path | first -%} {%- endif -%} {%- unless block -%} {%- assign copy_path = "_copy/" | append: include.name | append: ".md" -%} {%- assign block = site.copy | where: "relative_path", copy_path | first -%} {%- endunless -%} {{ block.content }}
